文件上传
要实现文件上传，你需要提供一个上传功能的实现。
export declare abstract class FileUploader {
    abstract uploadFile(type: string): string | Promise<string>;
}

import { FileUploader, Editor } from '@textbus/xnote'

const fileUploader: FileUploader = {
  uploadFile(type) {
    if (type === 'image') {
      const fileInput = document.createElement('input')
      fileInput.setAttribute('type', 'file')
      fileInput.setAttribute('accept', 'image/png, image/gif, image/jpeg, image/bmp, image/x-icon')
      // fileInput.multiple = uploadConfig.multiple
      fileInput.style.cssText = 'position: absolute; left: -9999px; top: -9999px; opacity: 0'
      const promise = new Promise(function (resolve) {
        fileInput.addEventListener('change', (event) => {
          const files = (event.target as HTMLInputElement).files!
          const file = files[0]
          const fileReader = new FileReader()

    }

    throw new Error(`没有实现 ${type} 类型的上传功能`)
  }
}
