协作编辑
Textbus 官方提供了基于 Yjs 的协作实现，要使用在线协作编辑，你需要先安装 Textbus 协作模块，并在 Textbus 实例中声明协作模块。
安装依赖
npm install @textbus/collaborate
添加协作模块
Textbus 官方提供了 y-websocket 和 hocuspocus 的实现，你也可以根据自己的需要，实现自己的协作连接器。
y-websocket 连接器示例：
import { YWebsocketConnector, CollaborateModule } from '@textbus/collaborate'
import { Textbus } from '@textbus/core'

// 创建协作模块
const collabModule = new CollaborateModule({
  createConnector(yDoc) {
    return new YWebsocketConnector('ws://example.com', 'docName', yDoc)
  }
})

const textbus = new Textbus({
  imports: [collabModule], // 添加协作模块到 Textbus
})

hocuspocus 连接器示例：
import { HocuspocusConnector, CollaborateModule } from '@textbus/collaborate'
import { Textbus } from '@textbus/core'

// 创建协作模块
const collabModule = new CollaborateModule({
  createConnector(yDoc) {
    return new HocuspocusConnector({
      url: 'ws://example.com',
      name: 'docName',
      document: yDoc
    })
  }
})

const textbus = new Textbus({
  imports: [collabModule], // 添加协作模块到 Textbus
 })

同步多用户光标
要实现多用户光标信息同步，需要实现 MessageBus 的特定方法。你可以在 MessageBus 中同步任意需要同步的信息，而不仅限于光标位置。
import { MessageBus, Message } from '@textbus/collaborate'
import { CollaborateCursor, UserSelectionCursor } from '@textbus/platform-browser'
import { Selection } from '@textbus/core'

const localUser = {
  username: '张三',
  color: '#ff0000',
  id: 'user_id',
}

class MyMessageBus extends MessageBus<UserSelectionCursor> {
  constructor(private selection: Selection,
              private collaborateCursor: CollaborateCursor) {
    super()
  }
  /**
   * 提供要共享到其它客户端的信息
   */
  get() {
    return {
      ...localUser,
      selection: this.selection.getPaths()
    }
  }
  /**
   * 当任意客户端信息有变更时，调用此方法
   */
  consume(message: Message<UserSelectionCursor>[]) {
    // 绘制多用户光标
    this.collaborateCursor.draw(message.filter(item => {
      // 排除当前本地用户
      return item.message.id !== localUser.id
    }).map(item => {
      return item.message
    }))
  }

}
const textbus = new Textbus({
  providers: [{
    // 提供协作光标通信类
    provide: MessageBus,
    useFactroy(selection: Selection, collaborateCursor: CollaborateCursor) {
      return new MyMessageBus(selection, collaborateCursor)
    },
    deps: [Selection, CollaborateCursor]
  }]
})

自定义连接器
要实现自定义的连接器，只需要实现以下接口即可：
/**
 * 协作通信通用接口
 */
export abstract class SyncConnector {
  /**
   * 当文档加载完成时触发的观察者
   */
  abstract onLoad: Observable<void>
  /**
   * 当文档 awareness 状态变更时触发的观察者
   */
  abstract onStateChange: Observable<any>

  /**
   * 设置 awareness 状态
   * @param key 状态的 key
   * @param data 状态值
   */
  abstract setLocalStateField(key: string, data: Record<string, any>): void

  /**
   * 当文档销毁时调用
   */
  abstract onDestroy(): void
}
